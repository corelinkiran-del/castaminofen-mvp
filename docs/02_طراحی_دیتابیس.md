### ۲.۱ USERS جدول

**مقصد:** ذخیره اطلاعات کاربران

```sql
CREATE TABLE users (
  -- Primary Key
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- Authentication
  email VARCHAR(255) UNIQUE NOT NULL,
  phone VARCHAR(20),
  password_hash VARCHAR(255) NOT NULL,
  
  -- Profile Information
  username VARCHAR(100) UNIQUE NOT NULL,
  first_name VARCHAR(100),
  last_name VARCHAR(100),
  avatar_url VARCHAR(500),
  bio TEXT,
  
  -- Preferences
  preferred_language VARCHAR(10) DEFAULT 'fa',
  preferred_timezone VARCHAR(50) DEFAULT 'Asia/Tehran',
  dark_mode BOOLEAN DEFAULT true,
  notification_enabled BOOLEAN DEFAULT true,
  
  -- Account Status
  subscription_status VARCHAR(50) DEFAULT 'free', 
  -- Options: free, premium, creator, admin
  subscription_end_date TIMESTAMP,
  is_creator BOOLEAN DEFAULT false,
  creator_status VARCHAR(50), 
  -- Options: pending, approved, rejected, suspended
  
  -- Verification
  email_verified BOOLEAN DEFAULT false,
  phone_verified BOOLEAN DEFAULT false,
  email_verified_at TIMESTAMP,
  phone_verified_at TIMESTAMP,
  
  -- Account Lifecycle
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  last_login_at TIMESTAMP,
  deleted_at TIMESTAMP, -- Soft delete
  
  -- Metadata
  device_count BIGINT DEFAULT 0,
  last_active_device VARCHAR(100),
  is_active BOOLEAN DEFAULT true
);

-- Indexes
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_subscription_status ON users(subscription_status);
CREATE INDEX idx_users_is_creator ON users(is_creator);
CREATE INDEX idx_users_created_at ON users(created_at);

-- Constraints
ALTER TABLE users ADD CONSTRAINT check_subscription_status 
  CHECK (subscription_status IN ('free', 'premium', 'creator', 'admin'));
ALTER TABLE users ADD CONSTRAINT check_creator_status 
  CHECK (creator_status IS NULL OR 
         creator_status IN ('pending', 'approved', 'rejected', 'suspended'));
```

**نکات مهم:**
- `id` از نوع UUID است برای privacy و distributed systems
- `password_hash` باید bcrypt یا argon2 باشد
- `deleted_at` برای soft deletes (GDPR compliance)
- `subscription_end_date` برای خودکار revoke کردن access

## ۳. نمایش‌های Materialized (Optional)

برای بهبود کارایی query های پر استفاده:

```sql
-- Top Creators View
CREATE MATERIALIZED VIEW top_creators_mv AS
SELECT 
  c.id,
  c.display_name,
  c.verified,
  c.total_followers,
  COUNT(ct.id) as total_contents,
  SUM(ct.total_plays) as total_plays
FROM creators c
LEFT JOIN contents ct ON c.id = ct.creator_id
GROUP BY c.id
ORDER BY c.total_followers DESC;

-- Trending Contents View
CREATE MATERIALIZED VIEW trending_contents_mv AS
SELECT 
  c.id,
  c.title,
  c.creator_id,
  c.type,
  c.category,
  COUNT(DISTINCT cp.user_id) as weekly_unique_listeners,
  SUM(cp.current_position / e.duration) as weighted_completion
FROM contents c
JOIN episodes e ON c.id = e.content_id
JOIN user_progress cp ON e.id = cp.episode_id
WHERE cp.played_at >= NOW() - INTERVAL '7 days'
GROUP BY c.id
ORDER BY weekly_unique_listeners DESC;

-- User Recommendations View (for Phase 2)
CREATE MATERIALIZED VIEW user_recommendations_mv AS
SELECT 
  u.id as user_id,
  c.id as recommended_content_id,
  COUNT(*) as relevance_score
FROM users u
JOIN user_progress up ON u.id = up.user_id
JOIN episodes e ON up.episode_id = e.id
JOIN contents c ON e.content_id = c.id
JOIN user_library ul ON u.id = ul.user_id AND ul.content_id = c.id
GROUP BY u.id, c.id;
```

## ۴. Stored Procedures

برای عملیات پیچیده:

```sql
-- Calculate user streaming statistics
CREATE OR REPLACE FUNCTION get_user_stats(p_user_id UUID)
RETURNS TABLE(
  total_playtime BIGINT,
  total_contents BIGINT,
  total_episodes BIGINT,
  completion_rate DECIMAL
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    COALESCE(SUM(up.current_position), 0)::BIGINT,
    COUNT(DISTINCT ul.content_id)::BIGINT,
    COUNT(DISTINCT up.episode_id)::BIGINT,
    COALESCE(AVG(CASE WHEN up.is_completed THEN 100 ELSE 0 END), 0)::DECIMAL
  FROM users u
  LEFT JOIN user_progress up ON u.id = up.user_id
  LEFT JOIN user_library ul ON u.id = ul.user_id
  WHERE u.id = p_user_id;
END;
$$ LANGUAGE plpgsql;

-- Archive old downloads
CREATE OR REPLACE PROCEDURE archive_expired_downloads()
LANGUAGE plpgsql
AS $$
BEGIN
  DELETE FROM downloads
  WHERE expires_at < NOW() AND auto_delete = true;
  
  RAISE NOTICE 'Expired downloads archived successfully';
END;
$$;
```

```yaml
Daily Backup:
  - Full backup: Once a day at 02:00 AM UTC
  - Incremental: Every 6 hours
  - Retention: 30 days
  
WAL (Write-Ahead Logging):
  - Continuous archiving
  - Retention: 7 days

  - Available for last 7 days
```
